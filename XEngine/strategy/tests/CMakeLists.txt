cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(XEngineExecutionTests LANGUAGES CXX)
 enable_testing()
endif()
find_package(Threads REQUIRED)
set(strategyProject "${CMAKE_CURRENT_LIST_DIR}/../..")
set(strategyPeer "${CMAKE_CURRENT_LIST_DIR}/../../../../XTrader/XTrader")
add_library(strategy_test_database STATIC MySqlTestDatabase.cpp)
target_compile_features(strategy_test_database PUBLIC cxx_std_20)
target_compile_definitions(strategy_test_database PRIVATE XENGINE_TEST_SCHEMA="${strategyProject}/exec/debug/sql/table_create.sql")
if(TARGET xengine_support)
 target_link_libraries(strategy_test_database PUBLIC xengine_support)
else()
 target_sources(strategy_test_database PRIVATE ../../database/CMySQL.cpp ../../database/IDataBase.cpp ../../database/common_db.cpp ../../common/utility.cpp)
 if(WIN32)
  target_include_directories(strategy_test_database PRIVATE "${strategyPeer}/include/debug/mysql")
  target_link_directories(strategy_test_database PUBLIC "${strategyPeer}/libs/debug")
  target_link_libraries(strategy_test_database PUBLIC libmysql)
  target_compile_options(strategy_test_database PRIVATE /utf-8 /W4)
 else()
  find_path(strategyMysqlInclude mysql/mysql.h REQUIRED)
  find_library(strategyMysqlLibrary mysqlclient REQUIRED)
  target_include_directories(strategy_test_database PRIVATE "${strategyMysqlInclude}")
  target_link_libraries(strategy_test_database PUBLIC "${strategyMysqlLibrary}")
 endif()
endif()
add_executable(strategy_execution_tests StrategyExecutionTests.cpp
 ../CStrategyExecution.cpp ../../trade/StrategyAccountView.cpp ../../trade/TradingTypes.cpp ../../request/MarketTypes.cpp)
target_compile_features(strategy_execution_tests PRIVATE cxx_std_20)
target_link_libraries(strategy_execution_tests PRIVATE strategy_test_database Threads::Threads)
if(MSVC)
 target_compile_options(strategy_execution_tests PRIVATE /utf-8 /W4 /WX)
else()
 target_compile_options(strategy_execution_tests PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME strategy_execution_tests COMMAND strategy_execution_tests)
set_tests_properties(strategy_execution_tests PROPERTIES TIMEOUT 120 SKIP_RETURN_CODE 77)

if(TARGET xengine_services)
 add_executable(strategy_behavior_tests StrategyBehaviorTests.cpp)
 target_link_libraries(strategy_behavior_tests PRIVATE xengine_services strategy_test_database)
elseif(WIN32)
 enable_language(C)
 set(strategyProject "${CMAKE_CURRENT_LIST_DIR}/../..")
 set(strategyPeer "${CMAKE_CURRENT_LIST_DIR}/../../../../XTrader/XTrader")
 include("${strategyProject}/indicator/IndicatorSources.cmake")
 add_library(strategy_test_indicators STATIC "${strategyProject}/indicator/CIndicator.cpp" ${indicatorVendorSources})

 add_executable(strategy_behavior_tests StrategyBehaviorTests.cpp
  ../strategies/CConfiguredStrategy.cpp ../strategies/CDualMaCrossStrategy.cpp
  ../CStrategyExecution.cpp ../../trade/StrategyAccountView.cpp ../../trade/TradingTypes.cpp
  ../../request/MarketTypes.cpp ../../request/v1/market.pb.cc)
 target_compile_features(strategy_behavior_tests PRIVATE cxx_std_20)

 target_include_directories(strategy_behavior_tests PRIVATE "${strategyProject}/request" "${strategyPeer}/include/debug/protobuf" "${strategyPeer}/include/debug/rapidjson")
 target_link_directories(strategy_behavior_tests PRIVATE "${strategyPeer}/libs/debug")
 target_link_libraries(strategy_behavior_tests PRIVATE strategy_test_indicators libprotobufd abseil_dll strategy_test_database)
 target_compile_options(strategy_behavior_tests PRIVATE /utf-8 /w)
endif()
