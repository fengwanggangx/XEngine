cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(XEngineExecutionTests LANGUAGES CXX)
 enable_testing()
endif()
find_package(Threads REQUIRED)
add_executable(strategy_execution_tests StrategyExecutionTests.cpp
 ../CStrategyExecution.cpp
 ../../trade/StrategyAccountView.cpp
 ../../trade/TradingTypes.cpp
 ../../request/MarketTypes.cpp)
target_compile_features(strategy_execution_tests PRIVATE cxx_std_20)
target_link_libraries(strategy_execution_tests PRIVATE Threads::Threads)
if(WIN32)
 target_link_libraries(strategy_execution_tests PRIVATE winsqlite3)
 if(MSVC)
  target_compile_options(strategy_execution_tests PRIVATE /utf-8 /W4 /WX)
 endif()
else()
 if(TARGET xengine_support)
  target_include_directories(strategy_execution_tests PRIVATE "${depInc}/sqlite")
  target_link_directories(strategy_execution_tests PRIVATE "${depLib}")
  target_link_libraries(strategy_execution_tests PRIVATE sqlite3)
 else()
  find_package(SQLite3 REQUIRED)
  target_link_libraries(strategy_execution_tests PRIVATE SQLite::SQLite3)
 endif()
 target_compile_options(strategy_execution_tests PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME strategy_execution_tests COMMAND strategy_execution_tests)
set_tests_properties(strategy_execution_tests PROPERTIES TIMEOUT 30)

if(TARGET xengine_services)
 add_executable(strategy_behavior_tests StrategyBehaviorTests.cpp)
 target_link_libraries(strategy_behavior_tests PRIVATE xengine_services)
elseif(WIN32)
 enable_language(C)
 set(strategyProject "${CMAKE_CURRENT_LIST_DIR}/../..")
 set(strategyPeer "${CMAKE_CURRENT_LIST_DIR}/../../../../XTrader/XTrader")
 include("${strategyProject}/indicator/IndicatorSources.cmake")
 add_library(strategy_test_indicators STATIC "${strategyProject}/indicator/CIndicator.cpp" ${indicatorVendorSources})
 target_include_directories(strategy_test_indicators PRIVATE ${indicatorVendorIncludes})
 target_compile_features(strategy_test_indicators PRIVATE cxx_std_20)
 set_property(TARGET strategy_test_indicators PROPERTY C_STANDARD 11)
 add_executable(strategy_behavior_tests StrategyBehaviorTests.cpp
  ../strategies/CConfiguredStrategy.cpp ../strategies/CDualMaCrossStrategy.cpp
  ../CStrategyExecution.cpp ../../trade/StrategyAccountView.cpp ../../trade/TradingTypes.cpp
  ../../request/MarketTypes.cpp ../../request/v1/market.pb.cc)
 target_compile_features(strategy_behavior_tests PRIVATE cxx_std_20)
 target_compile_definitions(strategy_behavior_tests PRIVATE PROTOBUF_USE_DLLS NOMINMAX)
 target_include_directories(strategy_behavior_tests PRIVATE "${strategyProject}/request" "${strategyPeer}/include/debug/protobuf" "${strategyPeer}/include/debug/rapidjson")
 target_link_directories(strategy_behavior_tests PRIVATE "${strategyPeer}/libs/debug")
 target_link_libraries(strategy_behavior_tests PRIVATE strategy_test_indicators libprotobufd abseil_dll winsqlite3)
 target_compile_options(strategy_behavior_tests PRIVATE /utf-8 /w)
endif()
if(TARGET strategy_behavior_tests)
 add_test(NAME strategy_behavior_tests COMMAND strategy_behavior_tests)
 set_tests_properties(strategy_behavior_tests PROPERTIES TIMEOUT 30)
endif()
