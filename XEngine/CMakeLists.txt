cmake_minimum_required(VERSION 3.20)
project(XEngine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
enable_testing()
 set(XENGINE_DEPS "${CMAKE_CURRENT_SOURCE_DIR}" CACHE PATH "XEngine dependency bundle")
 set(XENGINE_MODE "debug" CACHE STRING "Dependency configuration: debug or release")
 set(depInc "${XENGINE_DEPS}/include/${XENGINE_MODE}")
 set(depLib "${XENGINE_DEPS}/libs/${XENGINE_MODE}")
 file(GLOB supportSources CONFIGURE_DEPENDS basic/*.cpp common/*.cpp database/*.cpp ini/*.cpp network/*.cpp thread/*.cpp)
 file(GLOB protobufSources CONFIGURE_DEPENDS request/*.cc request/v1/*.cc)
 file(GLOB abslLibraries "${depLib}/libabsl_*.so")
 add_library(xengine_support STATIC ${supportSources} ${protobufSources} request/request.cpp request/RequestCenter.cpp request/MarketTypes.cpp system/CSession.cpp system/CHostMgr.cpp)
 target_include_directories(xengine_support PUBLIC "${depInc}/protobuf" "${depInc}/libevent" "${depInc}/mysql" "${depInc}/mysql/mysql" "${depInc}/sqlite" "${depInc}/rapidjson" "${depInc}/ini" request)
 target_link_directories(xengine_support PUBLIC "${depLib}")
 set(protoSuffix "")
 if(XENGINE_MODE STREQUAL "debug")
  set(protoSuffix "d")
 endif()
 target_link_libraries(xengine_support PUBLIC protobuf${protoSuffix} ${abslLibraries} utf8_validity utf8_range mysqlclient sqlite3 event event_core event_extra event_openssl event_pthreads Threads::Threads)

file(GLOB strategySources CONFIGURE_DEPENDS strategy/*.cpp strategy/strategies/*.cpp)
file(GLOB tradingSources CONFIGURE_DEPENDS trade/*.cpp)
add_library(xengine_trading STATIC ${tradingSources})
target_link_libraries(xengine_trading PUBLIC xengine_support)
add_library(xengine_services STATIC system/CEngineService.cpp system/CSubscriptionMgr.cpp ${strategySources})
target_link_libraries(xengine_services PUBLIC xengine_trading)
add_executable(XEngine main.cpp system/CBootLoader.cpp)
target_link_libraries(XEngine PRIVATE xengine_services)
add_executable(trading_protocol_tests trade/tests/TradeProtocolTests.cpp)
target_link_libraries(trading_protocol_tests PRIVATE xengine_trading)
target_include_directories(trading_protocol_tests PRIVATE trade)
add_test(NAME trading_protocol_tests COMMAND trading_protocol_tests)

set(CMAKE_BUILD_RPATH "${depLib}")

add_executable(broker_client_probe trade/tests/BrokerClientProbe.cpp)
target_link_libraries(broker_client_probe PRIVATE xengine_trading)

add_executable(broker_client_tests trade/tests/BrokerClientTests.cpp)
target_link_libraries(broker_client_tests PRIVATE xengine_trading)
target_include_directories(broker_client_tests PRIVATE trade)
add_test(NAME broker_client_tests COMMAND broker_client_tests)

add_executable(service_routing_tests strategy/tests/ServiceRoutingTests.cpp)
target_link_libraries(service_routing_tests PRIVATE xengine_services)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/service-routing-runtime/ini")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/service-routing-runtime/ini/system.ini" "[Broker]\n")
add_test(NAME service_routing_tests COMMAND service_routing_tests)
set_tests_properties(service_routing_tests PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/service-routing-runtime" TIMEOUT 30)
